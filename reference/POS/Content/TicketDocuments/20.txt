<script type="text/javascript">
    $(document).ready(function () {

        //$("#jqxgridLookupitems").jqxGrid('clearfilters');

        var url = '@Url.Action("getItemSelectAllAdvance", "Team")';

        var sourcepkl =
        {
            datatype: "json",
            datafields:
            [
                { name: 'TeamId', type: 'int' },
                { name: 'TeamName', type: 'string' },
            ],
            id: 'TeamId',
            url: url,
            cache: false,
            filter: function () {
                // update the grid and send a request to the server.
                $("#jqxgridLookupitems").jqxGrid('updatebounddata', 'filter');
            },
            sort: function () {
                // update the grid and send a request to the server.
                $("#jqxgridLookupitems").jqxGrid('updatebounddata', 'sort');
            },
            root: 'Rows',
            beforeprocessing: function (data) {
                if (data != null) {
                    sourcepkl.totalrecords = data[0].TotalRows;
                }
            }
        };
        var dataAdapterao = new $.jqx.dataAdapter(sourcepkl, {
            loadError: function (xhr, status, error) {
                alert(error);
            }
        });
        // initialize jqxGrid
        $("#jqxgridLookupitems").jqxGrid(
        {
            width: '100%',
            height: 400,
            source: dataAdapterao,
            filterable: true,
            showfilterrow: true,
            sortable: true,
            autoheight: false,
            pageable: true,
            virtualmode: true,
            rendergridrows: function (obj) {
                return obj.data;
            },
            rendertoolbar: function (toolbar) {
                $("#jqxgridLookupitems").dblclick(function () {
                    var selectedrowindex = $("#jqxgridLookupitems").jqxGrid('getselectedrowindex');
                    var dataRecord = $("#jqxgridLookupitems").jqxGrid('getrowdata', selectedrowindex);
                    if (dataRecord.TeamId > 0)
                    {
                        $dialog.dialog('close');

                        $('#TeamId').val(dataRecord.TeamId);
                        $('#TeamName').val(dataRecord.TeamName);
                        $('#btnSaveTeam').prop('disabled', false);
                        $('#btnSaveTeam').html("SAVE");
                        $('#btnSaveTeam').css('width', '95px');
                        $('#DeleteTeam').prop('disabled', false);
                        $('#TeamName').prop('disabled', false);
                    }
                });
            },
            columns: [
                    {
                        text: 'Sr.#', sortable: false, filterable: false, editable: false,
                        groupable: false, draggable: false, resizable: false,
                        datafield: '', columntype: 'number', width: 40,
                        cellsrenderer: function (row, column, value) {
                            return "<div style='margin:4px;'>" + (value + 1) + "</div>";
                        }
                    },

          { text: 'Team Name', datafield: 'TeamName', filtertype: 'input', width: 690 },
            ]
        });

    });
</script>
